import json
from datetime import datetime, timezone
from pathlib import Path

root = Path('.artifacts/ci-run')
target = Path('.trellis/tasks/11-17-github-native-build/research/remote-validation.json')
previous = json.loads(target.read_text())
run = json.loads((root / 'run-3.json').read_text())
jobs = json.loads((root / 'jobs-3.json').read_text())['jobs']
artifacts = json.loads((root / 'artifacts-3.json').read_text())['artifacts']
verified = {item['id']: item for item in json.loads((root / 'verified-artifacts-3.json').read_text())}
if previous['runId'] == run['id']:
    prior = previous['previousAttempt']
else:
    prior = {'runId': previous['runId'], 'url': previous['url'], 'commit': previous['commit'],
             'status': previous['status'], 'conclusion': previous['conclusion'],
             'jobs': [{'name': job['name'], 'conclusion': job['conclusion']} for job in previous['jobs']],
             'verifiedArtifacts': previous['artifacts'],
             'supersededReason': 'Explicit Apple deployment targets added; queued Apple jobs cancelled by concurrency policy.'}
report = {'observedAtUtc': datetime.now(timezone.utc).isoformat(), 'runId': run['id'],
          'url': run['html_url'], 'commit': run['head_sha'], 'branch': run['head_branch'],
          'status': run['status'], 'conclusion': run['conclusion'], 'event': run['event'],
          'jobs': [{'id': job['id'], 'name': job['name'], 'status': job['status'],
                    'conclusion': job['conclusion'], 'url': job['html_url'],
                    'createdAt': job['created_at'], 'runnerAssigned': bool(job['runner_id']),
                    'steps': [{'name': step['name'], 'conclusion': step['conclusion']}
                              for step in job.get('steps', [])]} for job in jobs],
          'artifacts': [], 'previousAttempt': prior,
          'limits': ['Existing native debugger tests fail; source check remains red.',
                     'No device execution, application packaging or distribution signing validation.',
                     'iOS device ZIP download encountered TLS EOF locally; remote job and artifact upload succeeded.',
                     'Queued jobs have no assigned runner and have not executed build scripts.',
                     'The macOS application project still declares 10.15; CI now targets 13.0.']}
for item in artifacts:
    proof = verified.get(item['id'])
    report['artifacts'].append({'id': item['id'], 'name': item['name'],
                               'bytes': item['size_in_bytes'], 'digest': item['digest'],
                               'expiresAt': item['expires_at'],
                               'url': f"{run['html_url']}/artifacts/{item['id']}",
                               'locallyVerified': proof is not None,
                               'localVerification': proof})
target.write_text(json.dumps(report, indent=2) + '\n')
print(f"Recorded run {run['id']}: {len(artifacts)} artifacts; {len(verified)} locally verified")
